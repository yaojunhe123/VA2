% Slide 2: Definition of progressive visualization
\begin{frame}{Progressive vs. Non-progressive Visualization}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      {\large\bfseries\color{ConferenceBlue}Non-progressive visualization\par}
      \vspace{0.55em}
      \small
      \begin{itemize}
        \setlength{\itemsep}{0.8em}
        \item Renders in one pass.
        \item Each change in the data, whether caused by interaction or new data, triggers a full rendering.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\large\bfseries\color{ConferenceBlue}Progressive visualization\par}
      \vspace{0.55em}
      \small
      \begin{itemize}
        \setlength{\itemsep}{0.65em}
        \item Rendering uses a series of passes that improve the result until completion.
        \item Each pass is limited to a requested latency and renders a meaningful partial result.
        \item Each partial result has a meaningful error or quality measure and an estimate of completion, such as a progress bar.
        \item Users can interact at any time; the visualization adapts as much as possible and can implement steering when needed.
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  \centering
  {\scriptsize\textcolor{gray}{Definition adapted from Ulmer et al., ``A Survey on Progressive Visualization'' (2024).}}
\end{frame}

\begin{frame}{Progressive Computation}
  \footnotesize
  \begin{tabularx}{0.96\textwidth}{@{}X X@{}}
    \textbf{$F_p$}: progressive function & \textbf{$P$}: function parameters; $D_i$: data at pass $i$ \\
    \textbf{$q$}: requested interval between outputs & \textbf{$t_i,m_i$}: time and memory at pass $i$ \\
    \textbf{$R_{i-1},R_i,R$}: previous, partial, and final result & \textbf{$R_0=\emptyset$}: first call has no previous result
  \end{tabularx}
  \vspace{0.3em}
  \centering
  \small
  \[
    F_p(P,D_i,q,R_{i-1})\xrightarrow{t_i,m_i}R_i,\qquad i=1,\ldots,z
  \]
  \[
    t_i\leq q,\qquad R_i\to R\;\text{or a termination criterion is reached.}
  \]
  \vspace{-0.35em}
  \includegraphics[width=0.64\textwidth]{assets/pezzotti-fig8-panels-only.png}\\[0.1em]
  {\centering\scriptsize\textcolor{gray}{Equations: Fekete \& Primet (2016) [1], as cited by Ulmer et al. (2024) [3].\\[-0.1em]
  Figure: Pezzotti et al. (2017) [2].}\par}
\end{frame}

% Review slide 2 of 6
\begin{frame}{Three Ways to Produce Progressive Results}
  \begin{center}
    \includegraphics[width=0.90\textwidth]{assets/ulmer-ppc-row.png}\\[-0.1em]
    {\tiny\textcolor{gray}{Ulmer et al. (2024), ``A Survey on Progressive Visualization.''}}
  \end{center}
  \vspace{0.35em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.30\textwidth}
      \raggedright
      {\small\bfseries\color{ConferenceBlue}Data chunking\par}
      \vspace{0.2em}
      {\centering\scriptsize\[
        \begin{gathered}
          D_i=d_1\cup\cdots\cup d_i\\[-0.1em]
          R_i=F_p(P,D_i,q,R_{i-1})
        \end{gathered}
      \]}
      \vspace{0.1em}
      \begin{minipage}[t][0.8cm][t]{\linewidth}
        \footnotesize $R_i$ summarizes the data processed so far. It may be incomplete for the full dataset.
      \end{minipage}
    \end{column}
    \begin{column}{0.30\textwidth}
      \raggedright
      {\small\bfseries\color{ConferenceBlue}Process chunking\par}
      \vspace{0.2em}
      {\centering\scriptsize\[
        \begin{gathered}
          D_i=D\\[-0.1em]
          R_i=F_p(P,D,q,R_{i-1})
        \end{gathered}
      \]}
      \vspace{0.1em}
      \begin{minipage}[t][0.8cm][t]{\linewidth}
        \footnotesize $R_i$ is the current iteration output on the full dataset.
      \end{minipage}
    \end{column}
    \begin{column}{0.30\textwidth}
      \raggedright
      {\small\bfseries\color{ConferenceBlue}Custom chunking\par}
      \vspace{0.2em}
      {\centering\scriptsize\[
        \begin{gathered}
          D_i=\pi_i(P,D,R_{i-1})\\[-0.1em]
          R_i=F_p(P,D_i,q,R_{i-1})
        \end{gathered}
      \]}
      \vspace{0.1em}
      \begin{minipage}[t][0.8cm][t]{\linewidth}
        \footnotesize $R_i$ follows a task-specific or hybrid schedule.
      \end{minipage}
    \end{column}
  \end{columns}
  \vfill
  {\scriptsize\textcolor{gray}{The custom-chunking formula is schematic. The survey gives no universal equation.}}
\end{frame}

% Conceptual framing: the sequence of intermediate states is itself analyzable
\begin{frame}{The Intermediate State $R_i$ as an Object of Analysis}
  \centering
  \vspace{0.15em}
  {\large
  $R_0\;\longrightarrow\;R_1\;\longrightarrow\;\cdots\;\longrightarrow\;R_i\;\longrightarrow\;\cdots\;\longrightarrow\;R_z$
  \par}
  \vspace{0.45em}
  {\normalsize The sequence $\{R_i\}$ exposes how an analysis develops over time.\par}
  \vspace{0.7em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \centering
      {\small\bfseries\color{ConferenceBlue}Pass-indexed\par}
      \vspace{0.25em}
      \small Each $R_i$ is a state at a particular pass $i$.
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering
      {\small\bfseries\color{ConferenceBlue}Partial\par}
      \vspace{0.25em}
      \small $R_i$ is an intermediate output; it need not yet be the terminal state $R_z$.
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering
      {\small\bfseries\color{ConferenceBlue}Evolving\par}
      \vspace{0.25em}
      \small Its relation to $R_{i-1}$ makes change across passes observable.
    \end{column}
  \end{columns}
  \vfill
  {\small\bfseries Candidate properties to examine:}
  {\small quality/error, uncertainty, change, stability, and convergence\par}
  {\scriptsize\textcolor{gray}{These properties require task-specific interpretation and measures. The analytical framing is this talk's; the progressive-visualization definition is from Ulmer et al. (2024).}}
\end{frame}
