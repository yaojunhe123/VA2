% Slide 2: Definition of progressive visualization
\begin{frame}{Progressive vs. Non-progressive Visualization}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      {\large\bfseries\color{ConferenceBlue}Non-progressive visualization\par}
      \vspace{0.55em}
      \small
      \begin{itemize}
        \setlength{\itemsep}{0.8em}
        \item Renders in one pass.
        \item Each change in the data, whether caused by interaction or new data, triggers a full rendering.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\large\bfseries\color{ConferenceBlue}Progressive visualization\par}
      \vspace{0.55em}
      \small
      \begin{itemize}
        \setlength{\itemsep}{0.65em}
        \item Rendering uses a series of passes that improve the result until completion.
        \item Each pass is limited to a requested latency and renders a meaningful partial result.
        \item Each partial result has a meaningful error or quality measure and an estimate of completion, such as a progress bar.
        \item Users can interact at any time; the visualization adapts as much as possible and can implement steering when needed.
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  \centering
  {\scriptsize\textcolor{gray}{Definition adapted from Ulmer et al., ``A Survey on Progressive Visualization'' (2024).}}
\end{frame}

\begin{frame}{Progressive Computation}
  \footnotesize
  \begin{tabularx}{0.96\textwidth}{@{}X X@{}}
    \textbf{$F_p$}: progressive function & \textbf{$P$}: function parameters; $D_i$: data at pass $i$ \\
    \textbf{$q$}: requested interval between outputs & \textbf{$t_i,m_i$}: time and memory at pass $i$ \\
    \textbf{$R_{i-1},R_i,R$}: previous, partial, and final result & \textbf{$R_0=\emptyset$}: first call has no previous result
  \end{tabularx}
  \vspace{0.3em}
  \centering
  \small
  \[
    F_p(P,D_i,q,R_{i-1})\xrightarrow{t_i,m_i}R_i,\qquad i=1,\ldots,z
  \]
  \[
    t_i\leq q,\qquad R_i\to R\;\text{or a termination criterion is reached.}
  \]
  \vspace{-0.35em}
  \includegraphics[width=0.64\textwidth]{assets/pezzotti-fig8-panels-only.png}\\[0.1em]
  {\centering\scriptsize\textcolor{gray}{Equations: Fekete \& Primet (2016) [1], as cited by Ulmer et al. (2024) [3].\\[-0.1em]
  Figure: Pezzotti et al. (2017) [2].}\par}
\end{frame}

\begin{frame}{Three Ways to Produce Progressive Results}
  \centering
  \includegraphics[width=0.94\textwidth]{assets/ulmer-ppc-row.png}\\[-0.1em]
  {\tiny\textcolor{gray}{Ulmer et al. (2024), ``A Survey on Progressive Visualization.''}}

    \begin{column}{0.32\textwidth}
      {\small\bfseries\color{ConferenceBlue}Data chunking\par}
      \vspace{0.2em}
      {\footnotesize Add data chunks successively. $R_i$ reflects the data processed so far.}
    \end{column}
    \begin{column}{0.32\textwidth}
      {\small\bfseries\color{ConferenceBlue}Process chunking\par}
      \vspace{0.2em}
      {\footnotesize Iterate over loaded data. $R_i$ is the algorithm's current output.}
    \end{column}
    \begin{column}{0.32\textwidth}
      {\small\bfseries\color{ConferenceBlue}Custom chunking\par}
      \vspace{0.2em}
      {\footnotesize Use a tailored or hybrid schedule beyond the two basic modes.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{A Taxonomy of Progressive Visualizations}
  \begin{itemize}
    \item \textbf{Visualization type:} temporal, geo-spatial, hierarchical, network, multidimensional, or field.
    \item \textbf{Processing:} data chunking, process chunking, or custom chunking.
    \item \textbf{Data domain:} known end or unknown end.
    \item \textbf{Visual update:} extend the current view or overwrite it with a new state.
  \end{itemize}
  \vfill
  {\scriptsize\textcolor{gray}{Source: Ulmer et al. (2024).}}
\end{frame}

% Review slide 4 of 6
\begin{frame}{Progressive Dimensionality Reduction}
  \begin{itemize}
    \item t-SNE builds an embedding through iterative optimization, but it must first compute high-dimensional similarities.
    \item Barnes-Hut t-SNE reduces optimization cost, yet initialization can still delay the first interpretable embedding.
    \item Q-SNE approximated neighbors for sparse document data; A-tSNE extends approximate similarities to dense high-dimensional data.
  \end{itemize}
  \vfill
  {\scriptsize\textcolor{gray}{Source: Pezzotti et al. (2017), Related Work.}}
\end{frame}

% Review slide 5 of 6
\begin{frame}{Field Trends and Open Challenges}
  \begin{itemize}
    \item The survey classifies 48 papers from 1999 to 2023; 33 of 48 use multidimensional visualization.
    \item Steering appears in 35 studies, uncertainty in 20, and real-time processing in 10.
    \item Open challenges include visual stability as estimates change, uncertainty communication, and evaluation beyond runtime.
  \end{itemize}
  \vfill
  {\scriptsize\textcolor{gray}{Source: Ulmer et al. (2024), study set and discussion.}}
\end{frame}

% Review slide 6 of 6
\begin{frame}{Synthesis and Paper Positioning}
  \begin{itemize}
    \item A-tSNE addresses an algorithm-level bottleneck: delayed initialization in high-dimensional t-SNE.
    \item The 2024 survey offers a field-level taxonomy and reviews evaluation methods and research gaps.
    \item \textbf{Synthesis:} progressive methods must make partial results timely, interpretable, steerable, and stable as they improve.
  \end{itemize}
  \vfill
  {\scriptsize\textcolor{gray}{Sources: Pezzotti et al. (2017); Ulmer et al. (2024).}}
\end{frame}
